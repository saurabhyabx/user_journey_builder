# Implementation Guide: AI-Powered User Journey Builder

## 🎯 Project Overview

Build an intelligent user journey mapping platform that:
1. **Conversational AI Interface** - Natural language chat to understand user's product vision
2. **Dual Input Methods** - Structured interview wizard OR free-form AI conversation
3. **AI-Generated Journey Maps** - Automatically creates visual user journey diagrams from conversation
4. **Intelligent Commenting** - Users can comment on any node, path, or workflow element for AI refinement
5. **Iterative Customization** - AI understands feedback and regenerates specific parts of the journey
6. **Version Control** - Track all changes, rollback to previous versions
7. **Lifecycle Mapping** - Complete customer journey: Entry → Customer → Recurring → Upgraded → Torchbearer
8. **Conditional Routing** - Smart branching (e.g., "expert" vs "beginner" paths)
9. **Contextual Interventions** - Offer services (hire expert, quotes) at strategic points
10. **Multi-Format Export** - Mermaid diagrams, PNG, JSON, or interactive prototypes

### Building From Scratch

This is a **100% original project** built from the ground up. We're not using anyone's codebase as a foundation.

```prisma
// Replaces Workflow - represents user journey
model Journey {
  id            String   @id @default(cuid())
  appProfileId  String
  appProfile    AppProfile @relation(fields: [appProfileId], references: [id], onDelete: Cascade)
  userId        String
  user          User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  
  name          String   @default("Untitled Journey")
  description   String?
  type          JourneyType @default(USER_FLOW)
  
  // Generated by AI

  nodes         JourneyNode[]
  connections   JourneyConnection[]

  updatedAt     DateTime @updatedAt
}

enum JourneyType {
  USER_FLOW        // Main user journey
  ONBOARDING       // Onboarding specific
  CONVERSION       // Purchase funnel
  RETENTION        // Re-engagement
  CUSTOM

// Replaces Node - represents journey touchpoint
model JourneyNode {
  id          String   @id @default(cuid())
  journeyId   String
  journey     Journey  @relation(fields: [journeyId], references: [id], onDelete: Cascade)
  
  name        String
  type        JourneyNodeType
  position    Json     // {x: number, y: number}
  
  // Node configuration
  data        Json     @default("{}")  // Flexible config per node type
  
  // Lifecycle stage
  lifecycleStage LifecycleStage?
  
  // Conditional routing
  conditions  Json?    // Rules for branching
  
  // Connections
  outgoing    JourneyConnection[] @relation("FromNode")
  incoming    JourneyConnection[] @relation("ToNode")
  
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt
}

  JOURNEY_START

  ONBOARDING_STEP

  ACTION_BUTTON

  SMS_SENT
  NOTIFICATION

  // Interventions

  UPSELL            // Upgrade prompt
  
  // Milestones
  CONVERSION        // Purchase, signup
  MILESTONE         // Achievement
  
  // Exit
  JOURNEY_END
}

  ENTRY              // First touchpoint
  PROSPECT           // Exploring
  CUSTOMER           // First purchase
  RECURRING          // Repeat customer
  UPGRADED           // Premium tier
  TORCHBEARER        // Advocate/referrer
  CHURNED            // Left
}

// Replaces Connection - represents journey path
model JourneyConnection {
  id          String   @id @default(cuid())
  journeyId   String
  journey     Journey  @relation(fields: [journeyId], references: [id], onDelete: Cascade)
  
  fromNodeId  String
  fromNode    JourneyNode @relation("FromNode", fields: [fromNodeId], references: [id], onDelete: Cascade)
  
  toNodeId    String

  // Analytics

  updatedAt   DateTime @updatedAt

  USER
  ASSISTANT
  SYSTEM
}

// Comments on workflow elements

  id          String   @id @default(cuid())

  user        User     @relation(fields: [userId], references: [id], onDelete: Cascade)

  aiResponse  String?  @db.Text  // AI's response to the comment
  
  // Threading
  parentId    String?

  replies     Comment[] @relation("CommentThread")
  
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt
  
  @@index([journeyId, targetId])
}

enum CommentTargetType {
  NODE
  CONNECTION
  JOURNEY
  PATH  // Specific path through journey
}

// Version control for journeys
model JourneyVersion {
  id          String   @id @default(cuid())
  journeyId   String
  journey     Journey  @relation(fields: [journeyId], references: [id], onDelete: Cascade)
  
  version     Int
  name        String?  // Optional version name
  description String?  // What changed
  
  // Snapshot of journey at this version
  snapshot    Json     // Complete journey state (nodes, connections)
  
  createdBy   String
  user        User     @relation(fields: [createdBy], references: [id])
  createdAt   DateTime @default(now())
  
  @@unique([journeyId, version])
  @@index([journeyId])
}

// Track what AI understood from user
model UserIntent {
  id          String   @id @default(cuid())
  userId      String
  user        User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  journeyId   String?
  journey     Journey? @relation(fields: [journeyId], references: [id], onDelete: Cascade)
  
  // Extracted intent
  intent      String   @db.Text  // Free-form description
  entities    Json     // Structured data: features, integrations, goals
  confidence  Float    @default(0.8)  // AI confidence in understanding
  
  // How AI interpreted it
  interpretation String @db.Text
  
  createdAt   DateTime @default(now())
}
```

```typescript
        where: { journeyId: input.journeyId },

    .input(z.object({

      return { success: true };

    .input(z.object({

      versionId2: z.string(),
```

```typescript
// src/lib/ai/prompts.ts

const JOURNEY_SYSTEM_PROMPT = `You are an expert UX architect. Generate user journey maps based on app profiles.

Output JSON with this structure:
{
  "nodes": [
    {
      "id": "unique-id",
      "type": "JOURNEY_START" | "ONBOARDING_STEP" | "DECISION_POINT" | etc,
      "name": "Step name",
      "position": { "x": 0, "y": 0 },
      "data": { /* node-specific config */ },
      "lifecycleStage": "ENTRY" | "PROSPECT" | etc (optional)
    }
  ],
  "connections": [
    {
      "from": "node-id",
      "to": "node-id",
      "label": "Yes" | "No" | "Skip" (optional),
      "condition": { /* conditional logic */ } (optional)
    }
  ]
}

Rules:
- Start with JOURNEY_START node
- Use DECISION_POINT for branching (e.g., expert vs beginner)
- Add OFFER_SERVICE nodes for interventions
- Tag key milestones with lifecycleStage
- Position nodes in logical flow (left to right, top to bottom)
`;

function buildJourneyPrompt(appProfile: AppProfile): string {
  return `Generate a user journey for this app:

App Name: ${appProfile.appName}
Type: ${appProfile.appType}
Target Users: ${appProfile.targetUsers}
Features: ${JSON.stringify(appProfile.features)}
Monetization: ${appProfile.monetization}

Special Requirements:
${appProfile.targetUsers === 'mixed' ? `
- At onboarding step 4, add a DECISION_POINT asking "Are you an expert or starting for the first time?"
- If "starting first time" → route to OFFER_SERVICE node (hire expert, get quote, skip option)
```

```typescript
// src/lib/export/mermaid.ts

export function generateMermaidDiagram(journey: Journey): string {
  let mermaid = 'graph TD\n';
  
  journey.nodes.forEach(node => {
    const shape = getNodeShape(node.type);
    mermaid += `  ${node.id}[${shape.start}${node.name}${shape.end}]\n`;
  });
  
  journey.connections.forEach(conn => {
    const label = conn.label ? `|${conn.label}|` : '';
    mermaid += `  ${conn.fromNodeId} ${label}--> ${conn.toNodeId}\n`;
  });
  
  return mermaid;
}

function getNodeShape(type: JourneyNodeType) {
  switch (type) {
    case 'JOURNEY_START':
      return { start: '([', end: '])' }; // Stadium shape
    case 'DECISION_POINT':
      return { start: '{', end: '}' }; // Diamond
    case 'CONVERSION':
      return { start: '[[', end: ']]' }; // Hexagon
    default:
      return { start: '[', end: ']' }; // Rectangle
  }
}
```

```typescript
// src/lib/quote-generator.ts

export function generateQuote(journey: Journey, appProfile: AppProfile): {
  basePrice: number;
  breakdown: Array<{ item: string; cost: number }>;
  total: number;
} {
  const breakdown = [];
  let total = 0;
  
  // Base cost by complexity
  const complexityCost = {
    simple: 5000,
    moderate: 15000,
    complex: 35000,
  }[appProfile.complexity || 'moderate'];
  
  breakdown.push({ item: 'Base development', cost: complexityCost });
  total += complexityCost;
  
  // Add per-node costs
  const nodeCount = journey.nodes.length;
  const nodeCost = nodeCount * 200;
  breakdown.push({ item: `${nodeCount} journey steps`, cost: nodeCost });
  total += nodeCost;
  
  // Add integration costs
  const integrations = appProfile.integrations as string[] || [];
  const integrationCost = integrations.length * 1500;
  if (integrationCost > 0) {
    breakdown.push({ item: `${integrations.length} integrations`, cost: integrationCost });
    total += integrationCost;
  }
  
  return { basePrice: complexityCost, breakdown, total };
}
```
